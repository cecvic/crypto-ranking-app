// Cron: Poll Birdeye for large DEX trades across all chains
// Schedule: Every 15 minutes (offset from poll-birdeye at :07, :22, :37, :52)
// Purpose: Fetch whale-sized trades ($100k+) for whale activity scoring

import { NextRequest, NextResponse } from 'next/server';
import { verifyCronRequestWithDevBypass } from '@/lib/qstash/verify';
import { getWhaleTradesByVolume, BirdeyeChain } from '@/lib/apis/birdeye';
import { getTokenAddresses } from '@/lib/db/birdeye-queries';
import { insertDexTrades } from '@/lib/db/dex-trade-queries';
import { checkRateLimit, RATE_LIMITS } from '@/lib/rate-limiter/distributed';

// Route config for extended timeout
export const maxDuration = 60; // 60 seconds max (Vercel free tier limit)

// Chains to poll in priority order (highest DEX volume first)
const CHAIN_PRIORITY: BirdeyeChain[] = [
  'solana',
  'ethereum',
  'base',
  'arbitrum',
  'bsc',
  'polygon',
  'optimism',
  'avalanche',
  'zksync',
  'sui',
  'aptos',
];

// Whale trade threshold in USD
const MIN_VOLUME_USD = 100_000; // $100k+ trades

// Tokens per chain to check for whale trades
const TOKENS_PER_CHAIN = 20; // Top 20 by volume per chain

async function handler(req: NextRequest): Promise<NextResponse> {
  const startTime = Date.now();
  console.log('[poll-whale-trades] Starting multi-chain whale trade polling...');

  const results: Record<string, { tokens: number; trades: number; error?: string }> = {};
  let totalTokens = 0;
  let totalTrades = 0;
  let chainsPolled = 0;

  try {
    // Check initial rate limit headroom (shares quota with poll-birdeye)
    const initialStatus = await checkRateLimit(RATE_LIMITS.BIRDEYE_ACCOUNT);
    const headroomPercent = (initialStatus.remaining / RATE_LIMITS.BIRDEYE_ACCOUNT.limit) * 100;
    console.log(`[poll-whale-trades] Initial rate limit headroom: ${headroomPercent.toFixed(1)}%`);

    // Skip if headroom is too low (prioritize price polling)
    if (headroomPercent < 20) {
      console.log('[poll-whale-trades] Rate limit headroom too low, skipping run');
      return NextResponse.json({
        success: false,
        error: 'Rate limit headroom too low',
        headroomPercent,
        resetAt: new Date(initialStatus.resetAt).toISOString(),
      }, { status: 429 });
    }

    // Reduce scope when headroom is tight
    let tokensPerChain = TOKENS_PER_CHAIN;
    let chainsToProcess = CHAIN_PRIORITY;
    if (headroomPercent < 40) {
      tokensPerChain = 10;
      chainsToProcess = CHAIN_PRIORITY.slice(0, 5); // Top 5 chains only
      console.log(`[poll-whale-trades] Reduced scope: ${chainsToProcess.length} chains, ${tokensPerChain} tokens/chain`);
    }

    // Poll chains sequentially to manage rate limits
    for (const chain of chainsToProcess) {
      // Check rate limit before each chain
      const rateCheck = await checkRateLimit(RATE_LIMITS.BIRDEYE_ACCOUNT);
      if (!rateCheck.allowed) {
        console.log(`[poll-whale-trades] Rate limit hit at chain ${chain}, stopping`);
        results[chain] = { tokens: 0, trades: 0, error: 'Rate limited' };
        break;
      }

      try {
        // Get top tokens by volume for this chain
        const addresses = await getTokenAddresses(chain, tokensPerChain);

        if (addresses.length === 0) {
          console.log(`[poll-whale-trades] No tokens registered for ${chain}, skipping`);
          results[chain] = { tokens: 0, trades: 0, error: 'No tokens registered' };
          continue;
        }

        let chainTrades = 0;

        // Fetch whale trades for each token
        for (const address of addresses) {
          const trades = await getWhaleTradesByVolume(chain, address, {
            minVolumeUsd: MIN_VOLUME_USD,
            limit: 50,
            txType: 'swap',
          });

          if (trades.length > 0) {
            // Insert trades into database
            const inserted = await insertDexTrades(
              trades.map(t => ({
                txHash: t.txHash,
                tokenAddress: t.tokenAddress,
                chain,
                blockTime: t.blockTime,
                side: t.side,
                volumeUsd: t.volumeUsd,
                priceUsd: t.priceUsd,
                tokenSymbol: t.tokenSymbol,
                source: t.source,
                poolId: t.poolId,
                dexName: t.source, // Use source as DEX name
              }))
            );
            chainTrades += inserted;
          }

          // Small delay between tokens to spread rate limit usage
          await new Promise(resolve => setTimeout(resolve, 50));
        }

        results[chain] = { tokens: addresses.length, trades: chainTrades };
        totalTokens += addresses.length;
        totalTrades += chainTrades;
        chainsPolled++;

        console.log(`[poll-whale-trades] ${chain}: ${chainTrades} trades from ${addresses.length} tokens`);

        // Delay between chains
        await new Promise(resolve => setTimeout(resolve, 100));

      } catch (error) {
        const errorMessage = error instanceof Error ? error.message : 'Unknown error';
        console.error(`[poll-whale-trades] Error polling ${chain}:`, error);
        results[chain] = { tokens: 0, trades: 0, error: errorMessage };
      }
    }

    const duration = Date.now() - startTime;
    console.log(`[poll-whale-trades] Completed in ${duration}ms: ${chainsPolled} chains, ${totalTrades} trades from ${totalTokens} tokens`);

    return NextResponse.json({
      success: true,
      chainsPolled,
      totalTokens,
      totalTrades,
      minVolumeUsd: MIN_VOLUME_USD,
      results,
      durationMs: duration,
      timestamp: new Date().toISOString(),
    });

  } catch (error) {
    console.error('[poll-whale-trades] Error:', error);

    return NextResponse.json({
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
      results,
      timestamp: new Date().toISOString(),
    }, { status: 500 });
  }
}

export const POST = verifyCronRequestWithDevBypass(handler);

// Allow GET for local testing
export async function GET(req: NextRequest) {
  if (process.env.NODE_ENV === 'production') {
    return NextResponse.json({ error: 'Method not allowed' }, { status: 405 });
  }
  return handler(req);
}

/**
 * Calculate whale score from combined Alchemy + DEX metrics
 */
function calculateWhaleScoreFromMetrics(
  metrics: WhaleMetrics,
  dexMetrics?: DexWhaleMetrics
): number {
  let whaleScore = 50; // Start neutral

  // Factor 1: Net flow direction (max +/-25 points)
  // Positive net flow (more leaving exchanges) = accumulation = bullish
  if (metrics.netFlow !== 0) {
    const flowImpact = Math.min(25, Math.abs(metrics.netFlow) / 10_000_000);
    whaleScore += metrics.netFlow > 0 ? flowImpact : -flowImpact;
  }

  // Factor 2: Transaction volume (max +/-15 points)
  // High whale activity = significant interest
  if (metrics.totalTransactions > 50) {
    whaleScore += 15;
  } else if (metrics.totalTransactions > 20) {
    whaleScore += 10;
  } else if (metrics.totalTransactions > 5) {
    whaleScore += 5;
  } else if (metrics.totalTransactions === 0) {
    // No activity - keep neutral or slightly bearish
    whaleScore -= 5;
  }

  // Factor 3: Accumulation ratio (max +/-10 points)
  const totalFlow = metrics.exchangeInflow + metrics.exchangeOutflow;
  if (totalFlow > 0) {
    const accumulationRatio = metrics.exchangeOutflow / totalFlow;
    if (accumulationRatio > 0.7) {
      whaleScore += 10; // Strong accumulation
    } else if (accumulationRatio > 0.55) {
      whaleScore += 5;
    } else if (accumulationRatio < 0.3) {
      whaleScore -= 10; // Strong distribution
    } else if (accumulationRatio < 0.45) {
      whaleScore -= 5;
    }
  }

  // NEW: Factor 4: DEX whale trade activity (max +/-15 points)
  if (dexMetrics) {
    // High volume of large trades indicates whale interest
    if (dexMetrics.largeTradeCount > 20) {
      whaleScore += 10;
    } else if (dexMetrics.largeTradeCount > 10) {
      whaleScore += 5;
    }

    // Buy/sell ratio indicates accumulation/distribution
    if (dexMetrics.buyVsSellRatio > 1.5) {
      whaleScore += 5; // Strong DEX accumulation
    } else if (dexMetrics.buyVsSellRatio < 0.67) {
      whaleScore -= 5; // Strong DEX distribution
    }
  }

  // Clamp to 0-100
  return Math.max(0, Math.min(100, whaleScore));
}

export async function getWhaleActivityFromDB(
  coinId: string,
  symbol: string
): Promise<WhaleActivity> {
  // Fetch both Alchemy and DEX metrics in parallel
  const [metrics, dexMetrics] = await Promise.all([
    getWhaleMetrics24h(coinId),
    getDexWhaleMetrics24h(coinId).catch(() => null),
  ]);

  // If no data from either source, fall back to existing logic
  if (metrics.totalTransactions === 0 && (!dexMetrics || dexMetrics.largeTradeCount === 0)) {
    return getWhaleActivity(coinId, symbol);
  }

  const whaleScore = calculateWhaleScoreFromMetrics(metrics, dexMetrics || undefined);

  return {
    coinId,
    largeTransactions24h: metrics.totalTransactions + (dexMetrics?.largeTradeCount || 0),
    exchangeInflow24h: metrics.exchangeInflow,
    exchangeOutflow24h: metrics.exchangeOutflow,
    netFlow24h: metrics.netFlow,
    whaleScore,
    source: 'alchemy+dex',
  };
}

/**
 * Batch version for efficient ranking computation
 */
export async function getBatchWhaleActivity(
  coins: Array<{ id: string; symbol: string }>
): Promise<Map<string, WhaleActivity>> {
  const coinIds = coins.map((c) => c.id);

  // Fetch both Alchemy and DEX metrics in parallel
  const [metricsMap, dexMetricsMap] = await Promise.all([
    getBatchWhaleMetrics(coinIds),
    getBatchDexWhaleMetrics(coinIds).catch(() => new Map<string, DexWhaleMetrics>()),
  ]);

  const results = new Map<string, WhaleActivity>();

  for (const coin of coins) {
    const metrics = metricsMap.get(coin.id);
    const dexMetrics = dexMetricsMap.get(coin.id);

    if ((!metrics || metrics.totalTransactions === 0) && (!dexMetrics || dexMetrics.largeTradeCount === 0)) {
      // No data from either source - return neutral score
      try {
        const fallback = await getWhaleActivity(coin.id, coin.symbol);
        results.set(coin.id, fallback);
      } catch {
        results.set(coin.id, {
          coinId: coin.id,
          largeTransactions24h: 0,
          exchangeInflow24h: 0,
          exchangeOutflow24h: 0,
          netFlow24h: 0,
          whaleScore: 50,
          source: 'defillama',
        });
      }
      continue;
    }

    const whaleScore = calculateWhaleScoreFromMetrics(
      metrics || { totalTransactions: 0, exchangeInflow: 0, exchangeOutflow: 0, walletToWallet: 0, netFlow: 0, avgTransactionSize: 0 },
      dexMetrics
    );

    results.set(coin.id, {
      coinId: coin.id,
      largeTransactions24h: (metrics?.totalTransactions || 0) + (dexMetrics?.largeTradeCount || 0),
      exchangeInflow24h: metrics?.exchangeInflow || 0,
      exchangeOutflow24h: metrics?.exchangeOutflow || 0,
      netFlow24h: metrics?.netFlow || 0,
      whaleScore,
      source: dexMetrics?.largeTradeCount ? 'alchemy+dex' : 'alchemy',
    });
  }

  return results;
}
