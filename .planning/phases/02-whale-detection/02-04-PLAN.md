// API: Get recent DEX whale trades and aggregate metrics
import { NextRequest, NextResponse } from 'next/server';
import { getRecentDexTrades, getDexWhaleMetrics24h } from '@/lib/db/dex-trade-queries';

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url);
    const coinGeckoId = searchParams.get('coinGeckoId');
    const chain = searchParams.get('chain');
    const limit = parseInt(searchParams.get('limit') || '50', 10);

    // Build filters
    const filters: {
      coinGeckoId?: string;
      chain?: string;
      minVolumeUsd?: number;
    } = {};

    if (coinGeckoId) filters.coinGeckoId = coinGeckoId;
    if (chain) filters.chain = chain;

    // Fetch recent trades
    const trades = await getRecentDexTrades(limit, filters);

    // Get aggregate metrics if filtering by coin
    let metrics = null;
    if (coinGeckoId) {
      metrics = await getDexWhaleMetrics24h(coinGeckoId);
    }

    // Calculate aggregate stats for all returned trades
    const aggregateStats = {
      totalTrades: trades.length,
      totalVolume: trades.reduce((sum, t) => sum + t.volumeUsd, 0),
      buyCount: trades.filter(t => t.side === 'buy').length,
      sellCount: trades.filter(t => t.side === 'sell').length,
      chains: [...new Set(trades.map(t => t.chain))],
    };

    return NextResponse.json({
      success: true,
      data: {
        trades: trades.map(t => ({
          id: t.id,
          txHash: t.txHash,
          tokenAddress: t.tokenAddress,
          tokenSymbol: t.tokenSymbol,
          chain: t.chain,
          side: t.side,
          volumeUsd: t.volumeUsd,
          priceUsd: t.priceUsd,
          timestamp: t.blockTimestamp.toISOString(),
          source: t.source,
          dexName: t.dexName,
        })),
        metrics,
        aggregateStats,
      },
      timestamp: new Date().toISOString(),
    });
  } catch (error) {
    console.error('[api/whale/dex-trades] Error:', error);
    return NextResponse.json(
      {
        success: false,
        error: error instanceof Error ? error.message : 'Unknown error',
      },
      { status: 500 }
    );
  }
}

  id: number;
  txHash: string;
  tokenAddress: string;

  side: 'buy' | 'sell';
  volumeUsd: number;

export interface DexWhaleMetrics {
  largeTradeCount: number;
  totalVolume: number;
  buyVolume: number;
  sellVolume: number;
  buyVsSellRatio: number;
  avgTradeSize: number;
}

interface DexTradesResponse {
  success: boolean;
  data: {
    trades: DexTrade[];
    metrics: DexWhaleMetrics | null;
    aggregateStats: {
      totalTrades: number;
      totalVolume: number;
      buyCount: number;
      sellCount: number;
      chains: string[];
    };
  };
  timestamp: string;
}

/**
 * Fetch recent DEX whale trades
 */
export function useDexTrades(options?: {
  coinGeckoId?: string;
  chain?: string;
  limit?: number;
}) {
  return useQuery({
    queryKey: ['dex-trades', options?.coinGeckoId, options?.chain, options?.limit],
    queryFn: async () => {
      const params = new URLSearchParams();
      if (options?.coinGeckoId) params.set('coinGeckoId', options.coinGeckoId);
      if (options?.chain) params.set('chain', options.chain);
      if (options?.limit) params.set('limit', options.limit.toString());

      const url = `/api/whale/dex-trades${params.toString() ? `?${params}` : ''}`;
      const response = await fetch(url);

      if (!response.ok) {
        throw new Error('Failed to fetch DEX trades');
      }

      const data: DexTradesResponse = await response.json();
      if (!data.success) {
        throw new Error('API returned error');
      }

      return data.data;
    },
    staleTime: 60_000, // 1 minute
    refetchInterval: 60_000, // Refetch every minute
  });
}

'use client';

import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Skeleton } from '@/components/ui/skeleton';
import { useDexTrades } from '@/hooks/use-whale-data';
import { ArrowUpIcon, ArrowDownIcon, ActivityIcon, TrendingUpIcon } from 'lucide-react';

interface DexWhaleSummaryProps {
  coinGeckoId?: string;
}

function formatVolume(volume: number): string {
  if (volume >= 1_000_000) {
    return `$${(volume / 1_000_000).toFixed(2)}M`;
  }
  if (volume >= 1_000) {
    return `$${(volume / 1_000).toFixed(1)}K`;
  }
  return `$${volume.toFixed(0)}`;
}

export function DexWhaleSummary({ coinGeckoId }: DexWhaleSummaryProps) {
  const { data, isLoading, error } = useDexTrades({
    coinGeckoId,
    limit: 100,
  });

  if (isLoading) {
    return <DexWhaleSummarySkeleton />;
  }

  if (error || !data) {
    return (
      <Card>
        <CardHeader>
          <CardTitle className="text-sm font-medium">DEX Whale Activity</CardTitle>
        </CardHeader>
        <CardContent>
          <p className="text-sm text-muted-foreground">Unable to load DEX whale data</p>
        </CardContent>
      </Card>
    );
  }

  const { aggregateStats, metrics } = data;
  const buyRatio = aggregateStats.totalTrades > 0
    ? (aggregateStats.buyCount / aggregateStats.totalTrades) * 100
    : 50;

  const sentiment = buyRatio > 60 ? 'bullish' : buyRatio < 40 ? 'bearish' : 'neutral';

  return (
    <Card>
      <CardHeader className="pb-2">
        <div className="flex items-center justify-between">
          <CardTitle className="text-sm font-medium flex items-center gap-2">
            <ActivityIcon className="h-4 w-4" />
            DEX Whale Activity (24h)
          </CardTitle>
          <Badge
            variant={
              sentiment === 'bullish'
                ? 'default'
                : sentiment === 'bearish'
                ? 'destructive'
                : 'secondary'
            }
          >
            {sentiment === 'bullish' ? (
              <TrendingUpIcon className="h-3 w-3 mr-1" />
            ) : null}
            {sentiment.charAt(0).toUpperCase() + sentiment.slice(1)}
          </Badge>
        </div>
      </CardHeader>
      <CardContent>
        <div className="grid grid-cols-2 gap-4">
          {/* Large Trades */}
          <div className="space-y-1">
            <p className="text-xs text-muted-foreground">Large Trades</p>
            <p className="text-2xl font-bold">{aggregateStats.totalTrades}</p>
            <p className="text-xs text-muted-foreground">
              {formatVolume(aggregateStats.totalVolume)} total
            </p>
          </div>

          {/* Buy/Sell Ratio */}
          <div className="space-y-1">
            <p className="text-xs text-muted-foreground">Buy/Sell Ratio</p>
            <div className="flex items-center gap-2">
              <div className="flex items-center text-green-500">
                <ArrowUpIcon className="h-4 w-4" />
                <span className="text-lg font-bold">{aggregateStats.buyCount}</span>
              </div>
              <span className="text-muted-foreground">/</span>
              <div className="flex items-center text-red-500">
                <ArrowDownIcon className="h-4 w-4" />
                <span className="text-lg font-bold">{aggregateStats.sellCount}</span>
              </div>
            </div>
            <p className="text-xs text-muted-foreground">
              {buyRatio.toFixed(0)}% buying pressure
            </p>
          </div>
        </div>

        {/* Chains active */}
        {aggregateStats.chains.length > 0 && (
          <div className="mt-4 pt-4 border-t">
            <p className="text-xs text-muted-foreground mb-2">Active on</p>
            <div className="flex flex-wrap gap-1">
              {aggregateStats.chains.map((chain) => (
                <Badge key={chain} variant="outline" className="text-xs">
                  {chain}
                </Badge>
              ))}
            </div>
          </div>
        )}

        {/* Metrics if available */}
        {metrics && (
          <div className="mt-4 pt-4 border-t grid grid-cols-2 gap-4">
            <div>
              <p className="text-xs text-muted-foreground">Avg Trade Size</p>
              <p className="font-medium">{formatVolume(metrics.avgTradeSize)}</p>
            </div>
            <div>
              <p className="text-xs text-muted-foreground">Buy/Sell Volume Ratio</p>
              <p className="font-medium">
                {metrics.buyVsSellRatio.toFixed(2)}x
                {metrics.buyVsSellRatio > 1.2 && (
                  <span className="text-green-500 ml-1 text-xs">accumulating</span>
                )}
                {metrics.buyVsSellRatio < 0.8 && (
                  <span className="text-red-500 ml-1 text-xs">distributing</span>
                )}
              </p>
            </div>
          </div>
        )}
      </CardContent>
    </Card>
  );
}

            <Skeleton className="h-3 w-20" />
            <Skeleton className="h-6 w-24" />
            <Skeleton className="h-3 w-20" />
          </div>
        </div>
      </CardContent>
    </Card>
  );
}
