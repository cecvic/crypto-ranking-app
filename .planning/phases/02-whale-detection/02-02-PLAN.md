// Database Query Functions for DEX Trade Tracking
import { getDb } from './client';
import { dexTrades, type NewDexTradeRow } from './schema';
import { getTokenMapping } from './whale-queries';
import { eq, and, gte, sql, inArray, desc } from 'drizzle-orm';

// ============================================
// INSERT OPERATIONS
// ============================================

/**
 * Batch insert DEX trades with automatic deduplication
 * Uses composite unique index on (transactionHash, tokenAddress, chain)
 */
export async function insertDexTrades(
  trades: Array<{
    txHash: string;
    tokenAddress: string;
    chain: string;
    blockTime: number;
    side: 'buy' | 'sell';
    volumeUsd: number;
    priceUsd: number;
    tokenSymbol?: string;
    coinGeckoId?: string;
    source: string;
    poolId?: string;
    dexName?: string;
  }>
): Promise<number> {
  if (trades.length === 0) return 0;

  // Map token addresses to CoinGecko IDs if not provided
  const addressesNeedingMapping = trades
    .filter(t => !t.coinGeckoId)
    .map(t => t.tokenAddress.toLowerCase());

  const uniqueAddresses = [...new Set(addressesNeedingMapping)];
  const mappingMap = new Map<string, string>();

  if (uniqueAddresses.length > 0) {
    const mappings = await Promise.all(
      uniqueAddresses.map(addr => getTokenMapping(addr))
    );
    mappings.forEach((m, i) => {
      if (m?.coinGeckoId) {
        mappingMap.set(uniqueAddresses[i], m.coinGeckoId);
      }
    });
  }

  // Prepare trades for insertion
  const values: NewDexTradeRow[] = trades.map(trade => ({
    transactionHash: trade.txHash,
    tokenAddress: trade.tokenAddress.toLowerCase(),
    chain: trade.chain,
    blockTimestamp: new Date(trade.blockTime * 1000),
    side: trade.side,
    volumeUsd: trade.volumeUsd.toString(),
    priceUsd: trade.priceUsd.toString(),
    tokenSymbol: trade.tokenSymbol || null,
    coinGeckoId: trade.coinGeckoId || mappingMap.get(trade.tokenAddress.toLowerCase()) || null,
    source: trade.source,
    poolId: trade.poolId || null,
    dexName: trade.dexName || null,
  }));

  // Batch insert with deduplication
  const result = await getDb()
    .insert(dexTrades)
    .values(values)
    .onConflictDoNothing()
    .returning({ id: dexTrades.id });

  console.log(`[dex-trades] Inserted ${result.length}/${trades.length} trades (deduped)`);
  return result.length;
}

export interface DexWhaleMetrics {
  largeTradeCount: number;
  totalVolume: number;
  buyVolume: number;
  sellVolume: number;
  buyVsSellRatio: number;
  avgTradeSize: number;
}

/**
 * Get DEX whale metrics for a specific coin (24h rolling window)
 */
export async function getDexWhaleMetrics24h(
  coinGeckoId: string
): Promise<DexWhaleMetrics> {
  const cutoff = new Date(Date.now() - 24 * 60 * 60 * 1000);

  const metrics = await getDb()
    .select({
      side: dexTrades.side,
      count: sql<number>`count(*)::int`,
      totalVolume: sql<number>`COALESCE(sum(${dexTrades.volumeUsd}::numeric), 0)::float`,
    })
    .from(dexTrades)
    .where(
      and(
        eq(dexTrades.coinGeckoId, coinGeckoId),
        gte(dexTrades.blockTimestamp, cutoff)
      )
    )
    .groupBy(dexTrades.side);

  const bySide = new Map(metrics.map(m => [m.side, m]));
  const buyVolume = bySide.get('buy')?.totalVolume || 0;
  const sellVolume = bySide.get('sell')?.totalVolume || 0;
  const buyCount = bySide.get('buy')?.count || 0;
  const sellCount = bySide.get('sell')?.count || 0;
  const totalCount = buyCount + sellCount;
  const totalVolume = buyVolume + sellVolume;

  return {
    largeTradeCount: totalCount,
    totalVolume,
    buyVolume,
    sellVolume,
    buyVsSellRatio: sellVolume > 0 ? buyVolume / sellVolume : (buyVolume > 0 ? 2 : 1),
    avgTradeSize: totalCount > 0 ? totalVolume / totalCount : 0,
  };
}

/**
 * Batch get DEX whale metrics for multiple coins (more efficient for ranking computation)
 */
export async function getBatchDexWhaleMetrics(
  coinGeckoIds: string[]
): Promise<Map<string, DexWhaleMetrics>> {
  const results = new Map<string, DexWhaleMetrics>();

  if (coinGeckoIds.length === 0) {
    return results;
  }

  const cutoff = new Date(Date.now() - 24 * 60 * 60 * 1000);

  const metrics = await getDb()
    .select({
      coinGeckoId: dexTrades.coinGeckoId,
      side: dexTrades.side,
      count: sql<number>`count(*)::int`,
      totalVolume: sql<number>`COALESCE(sum(${dexTrades.volumeUsd}::numeric), 0)::float`,
    })
    .from(dexTrades)
    .where(
      and(
        inArray(dexTrades.coinGeckoId, coinGeckoIds),
        gte(dexTrades.blockTimestamp, cutoff)
      )
    )
    .groupBy(dexTrades.coinGeckoId, dexTrades.side);

  // Group by coinGeckoId
  const byCoin = new Map<string, typeof metrics>();
  for (const m of metrics) {
    if (!m.coinGeckoId) continue;
    if (!byCoin.has(m.coinGeckoId)) {
      byCoin.set(m.coinGeckoId, []);
    }
    byCoin.get(m.coinGeckoId)!.push(m);
  }

  // Process each coin
  for (const coinId of coinGeckoIds) {
    const coinMetrics = byCoin.get(coinId) || [];
    const bySide = new Map(coinMetrics.map(m => [m.side, m]));

    const buyVolume = bySide.get('buy')?.totalVolume || 0;
    const sellVolume = bySide.get('sell')?.totalVolume || 0;
    const buyCount = bySide.get('buy')?.count || 0;
    const sellCount = bySide.get('sell')?.count || 0;
    const totalCount = buyCount + sellCount;
    const totalVolume = buyVolume + sellVolume;

    results.set(coinId, {
      largeTradeCount: totalCount,
      totalVolume,
      buyVolume,
      sellVolume,
      buyVsSellRatio: sellVolume > 0 ? buyVolume / sellVolume : (buyVolume > 0 ? 2 : 1),
      avgTradeSize: totalCount > 0 ? totalVolume / totalCount : 0,
    });
  }

  return results;
}

/**
 * Get recent DEX trades with optional filters
 */
export async function getRecentDexTrades(
  limit: number = 50,
  filters?: {
    coinGeckoId?: string;
    chain?: string;
    side?: 'buy' | 'sell';
    minVolumeUsd?: number;
  }
): Promise<Array<{
  id: number;
  txHash: string;
  tokenAddress: string;
  tokenSymbol: string | null;
  chain: string;
  side: string;
  volumeUsd: number;
  priceUsd: number;
  blockTimestamp: Date;
  source: string;
  dexName: string | null;
}>> {
  const conditions = [];

  if (filters?.coinGeckoId) {
    conditions.push(eq(dexTrades.coinGeckoId, filters.coinGeckoId));
  }
  if (filters?.chain) {
    conditions.push(eq(dexTrades.chain, filters.chain));
  }
  if (filters?.side) {
    conditions.push(eq(dexTrades.side, filters.side));
  }
  if (filters?.minVolumeUsd) {
    conditions.push(gte(dexTrades.volumeUsd, filters.minVolumeUsd.toString()));
  }

  const query = getDb()
    .select({
      id: dexTrades.id,
      txHash: dexTrades.transactionHash,
      tokenAddress: dexTrades.tokenAddress,
      tokenSymbol: dexTrades.tokenSymbol,
      chain: dexTrades.chain,
      side: dexTrades.side,
      volumeUsd: sql<number>`${dexTrades.volumeUsd}::float`,
      priceUsd: sql<number>`${dexTrades.priceUsd}::float`,
      blockTimestamp: dexTrades.blockTimestamp,
      source: dexTrades.source,
      dexName: dexTrades.dexName,
    })
    .from(dexTrades)
    .orderBy(desc(dexTrades.blockTimestamp))
    .limit(limit);

  if (conditions.length > 0) {
    return query.where(and(...conditions));
  }

  return query;
}

// ============================================
// CLEANUP OPERATIONS
// ============================================

/**
 * Delete DEX trades older than retention period
 */
export async function cleanupOldDexTrades(
  retentionDays: number = 30
): Promise<number> {
  const cutoff = new Date(Date.now() - retentionDays * 24 * 60 * 60 * 1000);

  const result = await getDb()
    .delete(dexTrades)
    .where(sql`${dexTrades.blockTimestamp} < ${cutoff}`)
    .returning({ id: dexTrades.id });

  console.log(`[dex-trades] Cleaned up ${result.length} trades older than ${retentionDays} days`);
  return result.length;
}
