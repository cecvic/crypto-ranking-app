export const dexTrades = pgTable('dex_trades', {
  id: serial('id').primaryKey(),

  // Trade identification (for deduplication)
  transactionHash: text('transaction_hash').notNull(),
  tokenAddress: text('token_address').notNull(),
  chain: text('chain').notNull(),

  // Trade details
  blockTimestamp: timestamp('block_timestamp', { withTimezone: true }).notNull(),
  side: text('side').notNull(), // 'buy' | 'sell'
  volumeUsd: decimal('volume_usd', { precision: 20, scale: 2 }).notNull(),
  priceUsd: decimal('price_usd', { precision: 20, scale: 8 }),

  // Token metadata (for aggregation)
  tokenSymbol: text('token_symbol'),
  coinGeckoId: text('coingecko_id'),

  // Source metadata
  source: text('source').notNull().default('birdeye-dex'),
  poolId: text('pool_id'),
  dexName: text('dex_name'),

  createdAt: timestamp('created_at', { withTimezone: true }).defaultNow(),
}, (table) => [
  // Prevent duplicate trades (txHash + tokenAddress + chain)
  uniqueIndex('idx_dex_trades_unique').on(table.transactionHash, table.tokenAddress, table.chain),
  // Query by coin for metrics
  index('idx_dex_trades_coin_time').on(table.coinGeckoId, table.blockTimestamp),
  // Query by time for cleanup
  index('idx_dex_trades_timestamp').on(table.blockTimestamp),
]);
```

interface BirdeyeTradesByVolumeResponse {
  success: boolean;
  data: {
    items: Array<{
      txHash: string;
      blockUnixTime: number;
      txType: 'swap' | 'add' | 'remove';
      owner: string;
      from: { address: string; symbol: string; decimals: number; amount: number };
      to: { address: string; symbol: string; decimals: number; amount: number };
      volumeUSD: number;
      source: string;
      poolId?: string;
    }>;
    hasNext: boolean;
  };
}

/**
 * Fetch large DEX trades for a token, sorted by volume (descending)
 * Uses /defi/v3/token/txs-by-volume endpoint
 * @param chain - Target chain
 * @param tokenAddress - Token contract address
 * @param options - Filter options (minVolumeUsd, limit, txType)
 */
export async function getWhaleTradesByVolume(
  chain: BirdeyeChain,
  tokenAddress: string,
  options?: {
    minVolumeUsd?: number;
    limit?: number;
    txType?: 'swap' | 'add' | 'remove' | 'all';
  }
): Promise<BirdeyeWhaleTrade[]> {
  try {
    console.log(`[Birdeye] Fetching whale trades for ${tokenAddress} on ${chain}...`);

    const response = await api.get<BirdeyeTradesByVolumeResponse>(
      '/defi/v3/token/txs-by-volume',
      {
        params: {
          address: tokenAddress,
          tx_type: options?.txType || 'swap',
          sort_type: 'desc',
          limit: Math.min(options?.limit || 50, 50), // API max is 50
          offset: 0,
        },
        headers: {
          'x-chain': chain,
        },
      }
    );

    if (!response.data?.success || !response.data?.data?.items) {
      console.log(`[Birdeye] No whale trades found for ${tokenAddress}`);
      return [];
    }

    // Filter by volume threshold if specified and map to BirdeyeWhaleTrade
    const minVolume = options?.minVolumeUsd || 0;
    const trades = response.data.data.items
      .filter(t => t.volumeUSD >= minVolume)
      .map(t => ({
        txHash: t.txHash,
        blockTime: t.blockUnixTime,
        source: t.source,
        // Determine side: if receiving the target token, it's a buy
        side: t.to.address.toLowerCase() === tokenAddress.toLowerCase() ? 'buy' as const : 'sell' as const,
        tokenAddress: tokenAddress,
        tokenSymbol: t.to.address.toLowerCase() === tokenAddress.toLowerCase() ? t.to.symbol : t.from.symbol,
        amount: t.to.address.toLowerCase() === tokenAddress.toLowerCase() ? t.to.amount : t.from.amount,
        priceUsd: t.volumeUSD / (t.to.address.toLowerCase() === tokenAddress.toLowerCase() ? t.to.amount : t.from.amount),
        volumeUsd: t.volumeUSD,
        poolId: t.poolId,
        owner: t.owner,
      }));

    console.log(`[Birdeye] Found ${trades.length} whale trades for ${tokenAddress} (min: $${minVolume})`);
    return trades;
  } catch (error) {
    console.error(`[Birdeye] Error fetching whale trades for ${tokenAddress}:`, error);
    return [];
  }
}
```
