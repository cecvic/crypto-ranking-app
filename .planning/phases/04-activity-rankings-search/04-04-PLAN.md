```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { useDebouncedValue } from 'use-debounce';

interface SearchToken {
  address: string;
  symbol: string;
  name: string;
  logoUri: string | null;
  price: number | null;
  liquidity: number | null;
  volume24h: number | null;
}

interface SearchResult {
  chain: string;
  tokens: SearchToken[];
}

interface UseTokenSearchOptions {
  enabled?: boolean;
  debounceMs?: number;
}

export function useTokenSearch(
  query: string,
  options: UseTokenSearchOptions = {}
) {
  const { enabled = true, debounceMs = 300 } = options;

  // Debounce the query
  const [debouncedQuery] = useDebouncedValue(query, debounceMs);

  return useQuery({
    queryKey: ['token-search', debouncedQuery],
    queryFn: async () => {
      const response = await fetch(
        `/api/tokens/search?q=${encodeURIComponent(debouncedQuery)}`
      );
      if (!response.ok) {
        throw new Error('Search failed');
      }
      const data = await response.json();
      return data.results as SearchResult[];
    },
    enabled: enabled && debouncedQuery.length >= 2,
    staleTime: 5 * 60 * 1000, // 5 minutes (match API cache)
    gcTime: 10 * 60 * 1000,
  });
}
```

- Use use-debounce for 300ms debounce
- Disable query if less than 2 characters
- Return typed results
- Match staleTime to API cache

Simple component to render a single search result:
```typescript
interface SearchResultItemProps {
  token: {
    address: string;
    symbol: string;
    name: string;
    logoUri: string | null;
    price: number | null;
    liquidity: number | null;
  };
  chain: string;
}

export function SearchResultItem({ token, chain }: SearchResultItemProps) {
  return (
    <div className="flex items-center gap-3 py-1">
      {/* Token icon or fallback */}
      {token.logoUri ? (
        <img src={token.logoUri} alt="" className="h-6 w-6 rounded-full" />
      ) : (
        <div className="h-6 w-6 rounded-full bg-muted flex items-center justify-center text-xs">
          {token.symbol.slice(0, 2)}
        </div>
      )}

      {/* Token info */}
      <div className="flex-1 min-w-0">
        <div className="flex items-center gap-2">
          <span className="font-medium">{token.symbol}</span>
          <span className="text-xs px-1.5 py-0.5 rounded bg-muted font-mono">
            {chain}
          </span>
        </div>
        <div className="text-sm text-muted-foreground truncate">
          {token.name}
        </div>
      </div>

      {/* Price if available */}
      {token.price && (
        <div className="text-sm font-mono">
          ${token.price < 0.01 ? token.price.toExponential(2) : token.price.toFixed(2)}
        </div>
      )}
    </div>
  );
}
```

**global-search.tsx:**
```typescript
'use client';

import { useEffect, useState, useCallback } from 'react';
import { useRouter } from 'next/navigation';
import {
  CommandDialog,
  CommandEmpty,
  CommandGroup,
  CommandInput,
  CommandItem,
  CommandList,
} from '@/components/ui/command';
import { useTokenSearch } from '@/hooks/use-token-search';
import { SearchResultItem } from './search-result-item';
import { SearchIcon } from 'lucide-react';

export function GlobalSearch() {
  const [open, setOpen] = useState(false);
  const [query, setQuery] = useState('');
  const router = useRouter();

  const { data: results, isLoading } = useTokenSearch(query, {
    enabled: open,
  });

  // Cmd+K / Ctrl+K keyboard shortcut
  useEffect(() => {
    const down = (e: KeyboardEvent) => {
      if (e.key === 'k' && (e.metaKey || e.ctrlKey)) {
        e.preventDefault();
        setOpen((open) => !open);
      }
    };

    document.addEventListener('keydown', down);
    return () => document.removeEventListener('keydown', down);
  }, []);

  // Reset query when dialog closes
  useEffect(() => {
    if (!open) {
      setQuery('');
    }
  }, [open]);

  const handleSelect = useCallback((chain: string, address: string) => {
    setOpen(false);
    // Navigate to token detail page (or tokens page filtered)
    router.push(`/dashboard/tokens?chain=${chain}&address=${address}`);
  }, [router]);

  const totalResults = results?.reduce((sum, g) => sum + g.tokens.length, 0) ?? 0;

  return (
    <CommandDialog open={open} onOpenChange={setOpen}>
      <CommandInput
        placeholder="Search tokens by name, symbol, or address..."
        value={query}
        onValueChange={setQuery}
      />
      <CommandList>
        <CommandEmpty>
          {isLoading ? (
            <div className="flex items-center justify-center py-6 text-muted-foreground">
              <SearchIcon className="mr-2 h-4 w-4 animate-pulse" />
              Searching...
            </div>
          ) : query.length < 2 ? (
            <span className="text-muted-foreground">
              Type at least 2 characters to search
            </span>
          ) : (
            <span>No tokens found for "{query}"</span>
          )}
        </CommandEmpty>

        {results?.map((group) => (
          <CommandGroup key={group.chain} heading={group.chain.toUpperCase()}>
            {group.tokens.map((token) => (
              <CommandItem
                key={`${group.chain}-${token.address}`}
                value={`${token.symbol} ${token.name} ${token.address}`}
                onSelect={() => handleSelect(group.chain, token.address)}
              >
                <SearchResultItem token={token} chain={group.chain} />
              </CommandItem>
            ))}
          </CommandGroup>
        ))}
      </CommandList>

      {totalResults > 0 && (
        <div className="border-t px-3 py-2 text-xs text-muted-foreground">
          {totalResults} result{totalResults !== 1 ? 's' : ''} across {results?.length} chain{results?.length !== 1 ? 's' : ''}
        </div>
      )}
    </CommandDialog>
  );
}
```
